module seven_seg_driver(
    input clk,
    input [7:0] val,        // 8-bit binary input (0-255)
    output reg [3:0] an,    // Anodes
    output reg [6:0] seg    // Cathodes
    );

    // BCD Conversion Logic (Double Dabble)
    reg [3:0] hundreds, tens, units;
    integer i;
    
    always @(*) begin
        hundreds = 4'd0;
        tens = 4'd0;
        units = 4'd0;
        
        for (i = 7; i >= 0; i = i - 1) begin
            if (hundreds >= 5) hundreds = hundreds + 3;
            if (tens >= 5) tens = tens + 3;
            if (units >= 5) units = units + 3;
            
            hundreds = {hundreds[2:0], tens[3]};
            tens = {tens[2:0], units[3]};
            units = {units[2:0], val[i]};
        end
    end

    // Time-multiplexing for 3 digits
    reg [19:0] refresh_counter; 
    always @(posedge clk) refresh_counter <= refresh_counter + 1;
    wire [1:0] active_digit = refresh_counter[19:18];

    reg [3:0] bcd_digit;
    always @(*) begin
        case(active_digit)
            2'b00: begin an = 4'b1110; bcd_digit = units; end    // Digit 0 (Right)
            2'b01: begin an = 4'b1101; bcd_digit = tens; end     // Digit 1 (Middle)
            2'b10: begin an = 4'b1011; bcd_digit = hundreds; end // Digit 2 (Left)
            default: begin an = 4'b1111; bcd_digit = 4'h0; end
        endcase
    end

    // Decimal to 7-segment decoder
    always @(*) begin
        case(bcd_digit)
            4'd0: seg = 7'b1000000; 4'd1: seg = 7'b1111001; 
            4'd2: seg = 7'b0100100; 4'd3: seg = 7'b0110000; 
            4'd4: seg = 7'b0011001; 4'd5: seg = 7'b0010010; 
            4'd6: seg = 7'b0000010; 4'd7: seg = 7'b1111000; 
            4'd8: seg = 7'b0000000; 4'd9: seg = 7'b0010000; 
            default: seg = 7'b1111111;
        endcase
    end
endmodule
